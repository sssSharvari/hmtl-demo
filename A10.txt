# ----------------------------------------
# Data Visualization III (Complete Solution)
# ----------------------------------------

import pandas as pd
import seaborn as sns
import matplotlib.pyplot as plt

# Load Iris dataset
df = sns.load_dataset("iris")

print("📊 Dataset Shape:", df.shape)
display(df.head())


# ----------------------------------------
# 1. FEATURES & TYPES
# ----------------------------------------

print("\n📌 Feature Classification:\n")

numeric_features = df.select_dtypes(include=['float64', 'int64']).columns
categorical_features = df.select_dtypes(include=['object']).columns

print("Numeric Features:", list(numeric_features))
print("Categorical Features:", list(categorical_features))


# ----------------------------------------
# 2. HISTOGRAMS (DISTRIBUTION)
# ----------------------------------------

df[numeric_features].hist(figsize=(10,8), bins=20)
plt.suptitle("Feature Distributions (Histograms)")
plt.show()


# ----------------------------------------
# 3. BOXPLOTS (OUTLIERS)
# ----------------------------------------

plt.figure(figsize=(10,6))
sns.boxplot(data=df[numeric_features])
plt.title("Boxplot of All Numeric Features")
plt.show()


# Feature-wise boxplots (with species comparison)
for col in numeric_features:
    plt.figure()
    sns.boxplot(x="species", y=col, data=df)
    plt.title(f"{col} Distribution by Species")
    plt.show()


# ----------------------------------------
# 4. OUTLIER DETECTION (IQR METHOD)
# ----------------------------------------

print("\n📌 Outlier Detection using IQR:\n")

for col in numeric_features:
    Q1 = df[col].quantile(0.25)
    Q3 = df[col].quantile(0.75)
    IQR = Q3 - Q1

    lower = Q1 - 1.5 * IQR
    upper = Q3 + 1.5 * IQR

    outliers = df[(df[col] < lower) | (df[col] > upper)]

    print(f"{col}: {len(outliers)} outliers")


# ----------------------------------------
# 5. INFERENCE (IMPORTANT FOR EXAM)
# ----------------------------------------

print("\n📊 Inference:")

print("""
1. All four features (sepal_length, sepal_width, petal_length, petal_width)
   are numeric, while species is categorical.

2. Histograms show:
   - Petal features have clear separation → strong classification power
   - Sepal features overlap more → less discriminative

3. Boxplots indicate:
   - Some outliers exist mainly in sepal_width
   - Petal features are more consistent

4. Overall:
   - Petal length & width are the most important features for distinguishing species.
""")
