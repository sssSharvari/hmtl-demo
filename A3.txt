import pandas as pd
import numpy as np
from IPython.display import display

# ----------------------------------------
# LOAD DATASET
# ----------------------------------------

# Replace path with your file location
df = pd.read_csv("E:/pic/New folder/Titanic-Dataset.csv")
df_original = df.copy()
print("Dataset Loaded Successfully\n")
display(df.head())


# ----------------------------------------
# CREATE CATEGORICAL VARIABLE (AGE GROUPS)
# ----------------------------------------
# Create Age Groups
bins = [0, 12, 18, 35, 60, 100]
labels = ["Child", "Teen", "Young ", "Adult", "Senior"]

df["Age_Group"] = pd.cut(df["Age"], bins=bins, labels=labels)

display(df[["Age", "Age_Group"]].head())



# ------------------------------------------------
# GROUPED SUMMARY STATISTICS (AGE GROUP vs FARE)
# ------------------------------------------------
grouped_stats = df.groupby("Age_Group")["Fare"].agg(
    Mean="mean",
    Median="median",
    Minimum="min",
    Maximum="max",
    Std_Dev="std"
).round(2)

print("Summary Statistics (Fare grouped by Age Group):")
display(grouped_stats)



# ------------------------------------------------
# CREATE LIST OF VALUES FOR EACH CATEGORY
# ------------------------------------------------
fare_lists = df.groupby("Age_Group")["Fare"].apply(list)

fare_df = fare_lists.reset_index(name="Fare Values")

display(fare_df)

summary = df.groupby("Age_Group")["Fare"].agg(
    Count="count",
    Sample=lambda x: list(x.head(3))
)

print("Fare Summary with Sample Values:")
display(summary)

import pandas as pd

url = "https://archive.ics.uci.edu/ml/machine-learning-databases/iris/iris.data"

columns = ["sepal_length", "sepal_width", "petal_length", "petal_width", "species"]

iris = pd.read_csv(url, header=None, names=columns)

iris.head()


#FILTER EACH SPECIES
setosa = iris[iris["species"] == "Iris-setosa"]
versicolor = iris[iris["species"] == "Iris-versicolor"]
virginica = iris[iris["species"] == "Iris-virginica"]


#FUNCTION FOR STATISTICS
def stats_summary(data, name):
    print(f"\nStatistics for {name}")
    display(data.describe(percentiles=[0.25, 0.5, 0.75]).round(2))

#APPLY FUNCTION
stats_summary(setosa, "Iris-setosa")
stats_summary(versicolor, "Iris-versicolor")
stats_summary(virginica, "Iris-virginica")
